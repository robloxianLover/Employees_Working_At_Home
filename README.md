Script para crear la base de datos con SQL en Postgres:
<br>
-- This script was generated by the ERD tool in pgAdmin 4.
-- Please log an issue at https://github.com/pgadmin-org/pgadmin4/issues/new/choose if you find any bugs, including reproduction steps.
BEGIN;


CREATE TABLE IF NOT EXISTS public.addresses
(
    address_id serial NOT NULL,
    line_1 character varying(150) COLLATE pg_catalog."default" NOT NULL,
    line_2 character varying(150) COLLATE pg_catalog."default",
    line_3 character varying(150) COLLATE pg_catalog."default",
    town_city character varying(100) COLLATE pg_catalog."default" NOT NULL,
    state_province character varying(100) COLLATE pg_catalog."default",
    country_code character varying(10) COLLATE pg_catalog."default" DEFAULT 'MX'::character varying,
    CONSTRAINT addresses_pkey PRIMARY KEY (address_id)
);

CREATE TABLE IF NOT EXISTS public.clients
(
    client_id serial NOT NULL,
    client_name character varying(150) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT clients_pkey PRIMARY KEY (client_id)
);

CREATE TABLE IF NOT EXISTS public.employee_on_projects
(
    employee_on_project_period_id serial NOT NULL,
    project_id integer NOT NULL,
    staff_id integer NOT NULL,
    from_day_date date NOT NULL,
    to_day_date date,
    hours_labored numeric(5, 2) DEFAULT 0,
    CONSTRAINT employee_on_projects_pkey PRIMARY KEY (employee_on_project_period_id)
);

CREATE TABLE IF NOT EXISTS public.employee_skills
(
    employee_id integer NOT NULL,
    skill_code character varying(10) COLLATE pg_catalog."default" NOT NULL,
    skill_level_code character varying(10) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT employee_skills_pkey PRIMARY KEY (employee_id, skill_code)
);

CREATE TABLE IF NOT EXISTS public.employees
(
    employee_id serial NOT NULL,
    role_code character varying(10) COLLATE pg_catalog."default",
    supervisor_id integer,
    first_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    last_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    email character varying(150) COLLATE pg_catalog."default" NOT NULL,
    salary numeric(10, 2) DEFAULT 0,
    CONSTRAINT employees_pkey PRIMARY KEY (employee_id),
    CONSTRAINT employees_email_key UNIQUE (email)
);

CREATE TABLE IF NOT EXISTS public.employees_addresses
(
    employee_id integer NOT NULL,
    address_id integer NOT NULL,
    date_address_from date NOT NULL DEFAULT CURRENT_DATE,
    date_address_to date,
    CONSTRAINT employees_addresses_pkey PRIMARY KEY (employee_id, address_id, date_address_from)
);

CREATE TABLE IF NOT EXISTS public.project_team
(
    project_id integer NOT NULL,
    employee_id integer NOT NULL,
    CONSTRAINT project_team_pkey PRIMARY KEY (project_id, employee_id)
);

CREATE TABLE IF NOT EXISTS public.projects
(
    project_id serial NOT NULL,
    client_id integer,
    project_name character varying(150) COLLATE pg_catalog."default" NOT NULL,
    project_start_date date DEFAULT CURRENT_DATE,
    project_end_date date,
    project_description text COLLATE pg_catalog."default",
    CONSTRAINT projects_pkey PRIMARY KEY (project_id)
);

CREATE TABLE IF NOT EXISTS public.ref_calendar
(
    day_date date NOT NULL,
    business_day_yn character(1) COLLATE pg_catalog."default" DEFAULT 'Y'::bpchar,
    day_number integer,
    period_id integer,
    CONSTRAINT ref_calendar_pkey PRIMARY KEY (day_date)
);

CREATE TABLE IF NOT EXISTS public.ref_roles
(
    role_code character varying(10) COLLATE pg_catalog."default" NOT NULL,
    role_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT ref_roles_pkey PRIMARY KEY (role_code)
);

CREATE TABLE IF NOT EXISTS public.ref_skill_levels
(
    skill_level_code character varying(10) COLLATE pg_catalog."default" NOT NULL,
    skill_level_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT ref_skill_levels_pkey PRIMARY KEY (skill_level_code)
);

CREATE TABLE IF NOT EXISTS public.ref_skills
(
    skill_code character varying(10) COLLATE pg_catalog."default" NOT NULL,
    skill_name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT ref_skills_pkey PRIMARY KEY (skill_code)
);

ALTER TABLE IF EXISTS public.employee_on_projects
    ADD CONSTRAINT fk_eop_employee FOREIGN KEY (staff_id)
    REFERENCES public.employees (employee_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employee_on_projects
    ADD CONSTRAINT fk_eop_from_date FOREIGN KEY (from_day_date)
    REFERENCES public.ref_calendar (day_date) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE NO ACTION;


ALTER TABLE IF EXISTS public.employee_on_projects
    ADD CONSTRAINT fk_eop_project FOREIGN KEY (project_id)
    REFERENCES public.projects (project_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employee_on_projects
    ADD CONSTRAINT fk_eop_to_date FOREIGN KEY (to_day_date)
    REFERENCES public.ref_calendar (day_date) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE NO ACTION;


ALTER TABLE IF EXISTS public.employee_skills
    ADD CONSTRAINT fk_emp_skill_emp FOREIGN KEY (employee_id)
    REFERENCES public.employees (employee_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employee_skills
    ADD CONSTRAINT fk_emp_skill_level FOREIGN KEY (skill_level_code)
    REFERENCES public.ref_skill_levels (skill_level_code) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employee_skills
    ADD CONSTRAINT fk_emp_skill_skill FOREIGN KEY (skill_code)
    REFERENCES public.ref_skills (skill_code) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employees
    ADD CONSTRAINT fk_role FOREIGN KEY (role_code)
    REFERENCES public.ref_roles (role_code) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE SET NULL;


ALTER TABLE IF EXISTS public.employees
    ADD CONSTRAINT fk_supervisor FOREIGN KEY (supervisor_id)
    REFERENCES public.employees (employee_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE SET NULL;


ALTER TABLE IF EXISTS public.employees_addresses
    ADD CONSTRAINT fk_emp_addr_addr FOREIGN KEY (address_id)
    REFERENCES public.addresses (address_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.employees_addresses
    ADD CONSTRAINT fk_emp_addr_emp FOREIGN KEY (employee_id)
    REFERENCES public.employees (employee_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.project_team
    ADD CONSTRAINT fk_pt_employee FOREIGN KEY (employee_id)
    REFERENCES public.employees (employee_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.project_team
    ADD CONSTRAINT fk_pt_project FOREIGN KEY (project_id)
    REFERENCES public.projects (project_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE CASCADE;


ALTER TABLE IF EXISTS public.projects
    ADD CONSTRAINT fk_project_client FOREIGN KEY (client_id)
    REFERENCES public.clients (client_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE SET NULL;

END;
